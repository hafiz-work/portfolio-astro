// Only Domestic General and Domestic TOU use this date-based protection.
// NEM and Non-Domestic calculators retain their existing thresholds.
const DOMESTIC_PROTECTION = {
  BASE_THRESHOLD: 600,
  EXPANDED_THRESHOLD: 800,
  START_DATE: "2026-09-01",
  END_DATE: "2026-12-31"
};

// Split usage and protection by calendar month. Unused protection in one month
// must NOT shelter taxable usage in another month (e.g. August/September 2026).
function getDomesticBillingPeriods(consumption, startDate, endDate) {
  const dayMs = 86400000;
  const start = Math.floor(new Date(startDate).getTime() / dayMs);
  const end = Math.floor(new Date(endDate).getTime() / dayMs);
  if (!Number.isFinite(start) || !Number.isFinite(end) || end < start) {
    throw new RangeError("A valid domestic billing start and end date is required.");
  }
  const totalDays = end - start + 1;
  const protectionStart = Date.parse(DOMESTIC_PROTECTION.START_DATE) / dayMs;
  const protectionEnd = Date.parse(DOMESTIC_PROTECTION.END_DATE) / dayMs;
  const periods = [];
  let allocatedUsage = 0;
  for (let cursor = start; cursor <= end;) {
    const date = new Date(cursor * dayMs);
    const nextMonth = Date.UTC(date.getUTCFullYear(), date.getUTCMonth() + 1, 1) / dayMs;
    const lastDay = Math.min(end, nextMonth - 1);
    const days = lastDay - cursor + 1;
    const threshold = cursor >= protectionStart && cursor <= protectionEnd
      ? DOMESTIC_PROTECTION.EXPANDED_THRESHOLD : DOMESTIC_PROTECTION.BASE_THRESHOLD;
    // Cumulative rounding preserves the entered kWh total, including half-kWh ties.
    const cumulativeUsage = lastDay === end ? consumption
      : Math.round(consumption * (lastDay - start + 1) / totalDays);
    const usage = cumulativeUsage - allocatedUsage;
    const allowance = Math.round(threshold * (lastDay - start + 1) / totalDays)
      - Math.round(threshold * (cursor - start) / totalDays);
    const nonST = Math.min(usage, allowance);
    periods.push({
      month: date.toISOString().slice(0, 7), days, totalDays,
      threshold, allowance, usage, nonST, st: usage - nonST,
      chargeable: consumption > threshold
    });
    allocatedUsage = cumulativeUsage;
    cursor = lastDay + 1;
  }
  return periods;
}

// Retain the existing aggregate-charge rounding when protection does not change.
// When the threshold changes, round each period's charge before adding it.
function getDomesticChargeGroups(periods) {
  if (periods.every(period => period.threshold === periods[0].threshold)) {
    return [{
      days: periods[0].totalDays, totalDays: periods[0].totalDays,
      usage: periods.reduce((sum, period) => sum + period.usage, 0),
      nonST: periods.reduce((sum, period) => sum + period.nonST, 0),
      st: periods.reduce((sum, period) => sum + period.st, 0)
    }];
  }
  return periods;
}

function roundDomesticAmount(value) {
  // Decimal half-away-from-zero rounding, including binary floating-point ties.
  return Math.sign(value) * Math.round((Math.abs(value) + Number.EPSILON * Math.max(1, Math.abs(value))) * 100) / 100;
}

function sumDomesticPeriodCharges(groups, usageKey, rate) {
  return roundDomesticAmount(groups.reduce((sum, group) =>
    sum + roundDomesticAmount(group[usageKey] * rate), 0));
}

function calculateDomesticRetailCharge(periods, amount) {
  const chargeableDays = periods.reduce((sum, period) =>
    sum + (period.chargeable ? period.days : 0), 0);
  return roundDomesticAmount(amount * chargeableDays / periods[0].totalDays);
}

function calculateDomesticPeriodAFA(periods, afaPrev, afa) {
  const zero = () => ({ rate: 0, nonST: 0, st: 0, total: 0 });
  const charges = periods.map((period, index) => {
    if (!period.chargeable) return zero();
    // The current UI supplies rates for the first and last billing months.
    // An intervening month (possible in a 31-day February cycle) uses history.
    const rate = Number(index === periods.length - 1 ? afa
      : index === 0 ? afaPrev : FUEL_BY_MONTH[period.month] || 0);
    return {
      rate,
      nonST: roundDomesticAmount(period.nonST * rate),
      st: roundDomesticAmount(period.st * rate),
      total: roundDomesticAmount(period.usage * rate)
    };
  });
  // Keep the existing two-row result interface; earlier months share its first row.
  const firstMonth = charges.slice(0, -1).reduce((sum, charge) => ({
    rate: charges[0].rate,
    nonST: roundDomesticAmount(sum.nonST + charge.nonST),
    st: roundDomesticAmount(sum.st + charge.st),
    total: roundDomesticAmount(sum.total + charge.total)
  }), zero());
  return { firstMonth, lastMonth: charges[charges.length - 1] };
}
