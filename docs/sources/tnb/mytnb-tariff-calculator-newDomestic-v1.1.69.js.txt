function calculateBill(type, consumption, afaPrev, afa) {
  //console.log(type, consumption)
  if (type === "newDomestic") {
    const startDate = new Date(document.getElementById("startDate").value);
    const endDate = new Date(document.getElementById("endDate").value);
    const periods = getDomesticBillingPeriods(consumption, startDate, endDate);
    const groups = getDomesticChargeGroups(periods);
    const nonSTConsumption = periods.reduce((sum, period) => sum + period.nonST, 0);
    const stConsumption = consumption - nonSTConsumption;
    const isAboveSTThreshold = stConsumption > 0;
    const isDifferent = periods.length > 1;

    const energyCharges = calculateTieredCharge(consumption, 'energy', groups);
    const fuelAdjustmentCharges = calculateDomesticPeriodAFA(periods, afaPrev, afa);
    const capacityCharges = calculateTieredCharge(consumption, 'capacity', groups);
    const networkCharges = calculateTieredCharge(consumption, 'network', groups);
    const retailRate = DOMESTIC.RETAIL_CHARGE_AMOUNT;
    const retailCharge = calculateDomesticRetailCharge(periods, retailRate);

    // Select incentive tier from the full bill usage, never from prorated usage.
    let totalEnergyEfficiencyIncentive = calculateEnergyEfficiencyIncentive(consumption);
    let energyEfficiencyIncentiveNonST = totalEnergyEfficiencyIncentive;
    let energyEfficiencyIncentiveST = 0;
    if (groups.length > 1) {
      const tier = DOMESTIC.INCENTIVE_TIERS.find(tier => consumption <= tier.max);
      const incentiveRate = tier ? tier.rate : 0;
      const incentiveFor = key => roundNum(groups.reduce((sum, group) =>
        sum - Math.round(Math.abs(group[key] * incentiveRate) * 100) / 100, 0));
      energyEfficiencyIncentiveNonST = incentiveFor('nonST');
      energyEfficiencyIncentiveST = incentiveFor('st');
      totalEnergyEfficiencyIncentive = roundNum(energyEfficiencyIncentiveNonST + energyEfficiencyIncentiveST);
    } else if (isAboveSTThreshold) {
      energyEfficiencyIncentiveNonST = totalEnergyEfficiencyIncentive * nonSTConsumption / consumption;
      energyEfficiencyIncentiveST = totalEnergyEfficiencyIncentive * stConsumption / consumption;
    }

    // Current Month Usage Charge calculations
    const currentMonthUsageChargeNonST = roundNum(
      energyCharges.nonST +
      fuelAdjustmentCharges.firstMonth.nonST +
      fuelAdjustmentCharges.lastMonth.nonST +
      capacityCharges.nonST +
      networkCharges.nonST +
      energyEfficiencyIncentiveNonST);

    const currentMonthUsageChargeST = roundNum(
      energyCharges.st +
      fuelAdjustmentCharges.firstMonth.st +
      fuelAdjustmentCharges.lastMonth.st +
      capacityCharges.st +
      networkCharges.st +
      retailCharge +
      energyEfficiencyIncentiveST);

    // Period line totals are rounded independently of the displayed tax splits.
    const currentMonthUsageCharge = groups.length > 1 ? roundNum(
      energyCharges.total + capacityCharges.total + networkCharges.total +
      fuelAdjustmentCharges.firstMonth.total + fuelAdjustmentCharges.lastMonth.total +
      retailCharge + totalEnergyEfficiencyIncentive
    ) : roundNum(currentMonthUsageChargeNonST + currentMonthUsageChargeST);

    // Service Tax (8%)
    const serviceTaxRate = periods.some(period => period.chargeable) ? 0.08 : 0;
    const serviceTax = serviceTaxRate > 0 ? roundNum(currentMonthUsageChargeST * serviceTaxRate) : 0;

    // KWTBB (1.6%)
    const kwtbbRate = consumption > 300 ? 0.016 : 0;
    const kwtbbBaseAmount =
      energyCharges.total +
      capacityCharges.total +
      networkCharges.total +
      totalEnergyEfficiencyIncentive;

    const kwtbb = roundNum(kwtbbBaseAmount * kwtbbRate);

    // Total Bill
    const totalBill = roundNum(currentMonthUsageCharge + serviceTax + kwtbb);

    const results = {
      consumption,
      consumptionNonST: nonSTConsumption,
      consumptionST: stConsumption,
      energyRate: energyCharges.rate,
      energyCharge: energyCharges.total,
      energyChargeNonST: energyCharges.nonST,
      energyChargeST: energyCharges.st,
      previousMonth: isDifferent,
      fuelAdjustmentRatePrev: fuelAdjustmentCharges.firstMonth.rate,
      fuelAdjustmentPrev: fuelAdjustmentCharges.firstMonth.total,
      fuelAdjustmentNonSTPrev: fuelAdjustmentCharges.firstMonth.nonST,
      fuelAdjustmentSTPrev: fuelAdjustmentCharges.firstMonth.st,
      fuelAdjustmentRate: fuelAdjustmentCharges.lastMonth.rate,
      fuelAdjustment: fuelAdjustmentCharges.lastMonth.total,
      fuelAdjustmentNonST: fuelAdjustmentCharges.lastMonth.nonST,
      fuelAdjustmentST: fuelAdjustmentCharges.lastMonth.st,
      capacityChargeRate: capacityCharges.rate,
      capacityCharge: capacityCharges.total,
      capacityChargeNonST: capacityCharges.nonST,
      capacityChargeST: capacityCharges.st,
      networkChargeRate: networkCharges.rate,
      networkCharge: networkCharges.total,
      networkChargeNonST: networkCharges.nonST,
      networkChargeST: networkCharges.st,
      retailRate,
      retailCharge,
      energyEfficiencyIncentive: totalEnergyEfficiencyIncentive,
      energyEfficiencyIncentiveNonST,
      energyEfficiencyIncentiveST,
      currentMonthUsageCharge,
      currentMonthUsageChargeNonST,
      currentMonthUsageChargeST,
      serviceTax,
      serviceTaxRate,
      kwtbb,
      kwtbbRate,
      totalBill,
    };

    fillTable(results);
  }

  return null;
}

function formatResultValue(value, decimal = true) {
  let num = Number(value);
  if (value === null || value === '' || isNaN(num) || !isFinite(num)) {
    return value;
  }
  num = Object.is(num, -0) ? 0 : num;
  return num.toLocaleString(undefined, {
    minimumFractionDigits: decimal ? 2 : 0,
    maximumFractionDigits: decimal ? 2 : 0
  });
}

function calculateTieredCharge(consumption, chargeType, groups) {
  const rate = consumption > DOMESTIC.HIGH_USAGE_THRESHOLD
    ? DOMESTIC.RATES.tier2[chargeType] : DOMESTIC.RATES.tier1[chargeType];
  const nonST = sumDomesticPeriodCharges(groups, 'nonST', rate);
  const st = sumDomesticPeriodCharges(groups, 'st', rate);
  const total = groups.length > 1 ? sumDomesticPeriodCharges(groups, 'usage', rate)
    : roundNum(nonST + st);
  return { rate, nonST, st, total };
}

function calculateFuelAdjustment(consumption, startDate, endDate, afaPrev, afa) {
  return calculateDomesticPeriodAFA(
    getDomesticBillingPeriods(consumption, startDate, endDate), afaPrev, afa);
}

function calculateEnergyEfficiencyIncentive(consumption) {
  if (consumption < 0) return 0;

  for (const tier of DOMESTIC.INCENTIVE_TIERS) {
    if (consumption <= tier.max) {
      //return consumption * tier.rate;
	  return -Math.round(Math.abs(consumption * tier.rate) * 100) / 100;
    }
  }
  return 0;
}

function roundNum(num) {
  return roundDomesticAmount(num);
}
